% Motivates the "greedy trap" (RQ3). Panel (b) is a NEUTRAL "globally coherent"
% alternative, deliberately not named after the future-work constrained beam search.
% Requires (in preamble): tikz (positioning, arrows.meta, backgrounds, calc),
%                         pifont, and the shared sv* colour palette.

\begin{figure}[!ht]
\centering
\begin{tikzpicture}[
    >=Stealth, font=\small,
    poi/.style={circle, draw=svBlue, line width=1pt, fill=svBlueT,
                minimum size=0.74cm, font=\footnotesize\bfseries, text=svInk},
    start/.style={poi, draw=svGreen, fill=svGreenT},
    bad/.style={poi, draw=svRed, fill=svRedT},
    dlabel/.style={font=\scriptsize, text=svGray},
    greedy/.style={->, line width=1pt, svRed, densely dashed},
    good/.style={->, line width=1pt, svBlue},
    panel/.style={rounded corners=8pt, fill=svGrayT},
    badge/.style={rounded corners=4pt, draw=svGray, line width=0.5pt,
                  fill=white, inner sep=4pt, font=\scriptsize},
]
% --- panel backgrounds ---
\begin{scope}[on background layer]
  \fill[panel] (-1.15,-1.25) rectangle (4.35,3.55);
  \fill[panel] ( 5.05,-1.25) rectangle (10.75,3.55);
\end{scope}

% ========== (a) Greedy decoding ==========
\node[font=\bfseries, text=svInk]            at (1.6,3.2)  {(a) Greedy decoding};
\node[font=\scriptsize\itshape, text=svRed]  at (1.6,2.78) {Pick the highest-scoring next POI};

\node[start] (gS) at (0.0,1.7)  {S};
\node[bad]   (gA) at (3.3,2.0)  {A};
\node[bad]   (gB) at (0.5,0.2)  {B};
\node[bad]   (gC) at (3.0,-0.1) {C};

\draw[greedy] (gS) -- (gA) node[midway, above, dlabel] {2.1\,km};
\draw[greedy] (gA) -- (gB) node[midway, above, sloped, dlabel] {3.0\,km};
\draw[greedy] (gB) -- (gC) node[midway, below, dlabel] {2.5\,km};

\node[badge] at (1.6,-0.9) {\textcolor{svRed}{\ding{55}}\;\, \textbf{7.6\,km} total $\cdot$ zigzag};

% ========== (b) Globally coherent route ==========
\node[font=\bfseries, text=svInk]             at (7.9,3.2)  {(b) Globally coherent route};
\node[font=\scriptsize\itshape, text=svBlue]  at (7.9,2.78) {Look beyond the next step};

\node[start] (cS) at (5.9,1.7)  {S};
\node[poi]   (cB) at (6.5,0.1)  {B};
\node[poi]   (cC) at (8.7,-0.1) {C};
\node[poi]   (cA) at (9.3,1.9)  {A};

\draw[good] (cS) -- (cB) node[midway, left, dlabel] {1.8\,km};
\draw[good] (cB) -- (cC) node[midway, below, dlabel] {2.0\,km};
\draw[good] (cC) -- (cA) node[midway, right, dlabel] {2.4\,km};

\node[badge] at (7.9,-0.9) {\textcolor{svGreen}{\ding{51}}\;\, \textbf{6.2\,km} total $\cdot$ coherent};

\end{tikzpicture}
\caption[The greedy trap in itinerary generation]{The \emph{greedy trap}: (a)~greedy decoding
yields a zigzag route, whereas (b)~looking beyond the next step visits the \emph{same} POIs in a
coherent order at lower travel cost.}
\label{fig:greedy-trap}
\end{figure}
